01 • SEARCHING ALGORITHM

🔎 Linear Search

#include <stdio.h>
int main() {
    int n, target, i, position = -1;
    scanf("%d", &n);
    int a[n];
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    scanf("%d", &target);
    for (i = 0; i < n; i++) {
        if (a[i] == target) {
            position = i;
            break;
        }
    }
    if (position != -1)
        printf("Element found at index %d\n", position);
    else
        printf("Element not found\n");
    return 0;
}

02 • SEARCHING ALGORITHM

⚡ Binary Search






💻 Iterative C Program







#include <stdio.h>
int main() {
    int n, target, low, high, mid, position = -1;
    scanf("%d", &n);
    int a[n];
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);
    scanf("%d", &target);
    low = 0;
    high = n - 1;
    while (low <= high) {
        mid = low + (high - low) / 2;
        if (a[mid] == target) {
            position = mid;
            break;
        } else if (target < a[mid]) {
            high = mid - 1;
        } else {
            low = mid + 1;
        }
    }
    return 0;
}

🔁 Recursive Binary Search







#include <stdio.h>
int binarySearch(int a[], int low, int high, int target) {
    if (low > high) return -1;
    int mid = low + (high - low) / 2;
    if (a[mid] == target) return mid;
    if (target < a[mid]) return binarySearch(a, low, mid - 1, target);
    return binarySearch(a, mid + 1, high, target);
}
