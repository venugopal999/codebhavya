🔍 Searching

Learn important searching techniques in C from basic linear search to interview-oriented advanced searching methods.

💡 Core Concept

Searching is the basic computing operation of finding the location of a target element within a dataset collection. The efficiency of a searching approach heavily relies on the underlying structure layout and whether the collection array sequence is pre-sorted or unsorted.

01 • SEARCHING ALGORITHM

🔎 Linear Search

Linear search scans each item sequentially from the very beginning of the dataset collection until a valid identity match matches the query parameters or the structure bounds are reached.

#include <stdio.h>

int main() {
    int n, target, i, position = -1;
    
    printf("Enter number of elements: ");
    scanf("%d", &n);
    
    int a[n];
    printf("Enter %d elements:\n", n);
    for (i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }
    
    printf("Enter element to search: ");
    scanf("%d", &target);
    
    for (i = 0; i < n; i++) {
        if (a[i] == target) {
            position = i;
            break;
        }
    }
    
    if (position != -1)
        printf("Element found at index %d\n", position);
    else
        printf("Element not found\n");
        
    return 0;
}

02 • SEARCHING ALGORITHM

⚡ Binary Search








📋 Prerequisites


Binary search strictly requires the data collection sequence array framework to be pre-sorted either in ascending or descending sequence. It follows a divide-and-conquer strategy, consistently cutting the active search space range half down each step.





💻 Iterative C Program







#include <stdio.h>

int main() {
    int n, target, low, high, mid, position = -1;
    
    printf("Enter number of elements (sorted): ");
    scanf("%d", &n);
    
    int a[n];
    printf("Enter %d sorted elements:\n", n);
    for (int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }
    
    printf("Enter element to search: ");
    scanf("%d", &target);
    
    low = 0;
    high = n - 1;
    
    while (low <= high) {
        mid = low + (high - low) / 2;
        
        if (a[mid] == target) {
            position = mid;
            break;
        } else if (target < a[mid]) {
            high = mid - 1;
        } else {
            low = mid + 1;
        }
    }
    
    if (position != -1)
        printf("Element found at index %d\n", position);
    else
        printf("Element not found\n");
        
    return 0;
}

🔁 Recursive Binary Search







#include <stdio.h>

int binarySearch(int a[], int low, int high, int target) {
    if (low > high) 
        return -1;
        
    int mid = low + (high - low) / 2;
    
    if (a[mid] == target) 
        return mid;
        
    if (target < a[mid]) 
        return binarySearch(a, low, mid - 1, target);
        
    return binarySearch(a, mid + 1, high, target);
}

int main() {
    int a[] = {3, 7, 11, 18, 25, 31, 40};
    int n = 7;
    int target = 31;
    int result = binarySearch(a, 0, n - 1, target);
    if (result != -1)
        printf("Element found at index %d\n", result);
    else
        printf("Element not found\n");
    return 0;
}

03 • ADVANCED SEARCH

🌀 Rotated Binary Search






Finds items effectively in a sorted layout vector structure that has been shifted or circular-rotated around a specific pivot index address offset point.







#include <stdio.h>

int searchRotated(int a[], int n, int target) {
    int low = 0, high = n - 1;
    
    while (low <= high) {
        int mid = low + (high - low) / 2;
        
        if (a[mid] == target) 
            return mid;
            
        if (a[low] <= a[mid]) {
            if (target >= a[low] && target < a[mid]) 
                high = mid - 1;
            else 
                low = mid + 1;
        } else {
            if (target > a[mid] && target <= a[high]) 
                low = mid + 1;
            else 
                high = mid - 1;
        }
    }
    return -1;
}

int main() {
    int a[] = {18, 25, 31, 40, 3, 7, 11};
    int n = 7;
    int target = 7;
    int idx = searchRotated(a, n, target);
    printf("Index of target: %d\n", idx);
    return 0;
}

04 • SEARCHING ALGORITHM

🦘 Jump Search






Jump search reduces comparison counts in sorted structures by skipping ahead across blocks of size $\sqrt{n}$ instead of iterating element-by-element.







#include <stdio.h>
#include <math.h>

int jumpSearch(int a[], int n, int target) {
    int step = sqrt(n);
    int prev = 0;
    
    while (a[(step < n ? step : n) - 1] < target) {
        prev = step;
        step += sqrt(n);
        if (prev >= n) 
            return -1;
    }
    
    while (a[prev] < target) {
        prev++;
        if (prev == (step < n ? step : n)) 
            return -1;
    }
    
    if (a[prev] == target) 
        return prev;
        
    return -1;
}

int main() {
    int a[] = {3, 7, 11, 18, 25, 31, 40, 45, 50};
    int n = 9;
    int target = 31;
    printf("Found at index: %d\n", jumpSearch(a, n, target));
    return 0;
}

05 • SEARCHING ALGORITHM

📈 Interpolation Search






An improvement over standard binary search for uniformly distributed values, finding items using proportional formula estimations.







#include <stdio.h>

int interpolationSearch(int a[], int n, int target) {
    int low = 0, high = n - 1;
    
    while (low <= high && target >= a[low] && target <= a[high]) {
        if (low == high) {
            if (a[low] == target) return low;
            return -1;
        }
        
        int pos = low + (((double)(high - low) / (a[high] - a[low])) * (target - a[low]));
        
        if (a[pos] == target) 
            return pos;
        if (a[pos] < target) 
            low = pos + 1;
        else 
            high = pos - 1;
    }
    return -1;
}

int main() {
    int a[] = {10, 20, 30, 40, 50, 60, 70};
    int n = 7;
    int target = 60;
    printf("Index match: %d\n", interpolationSearch(a, n, target));
    return 0;
}

06 • SEARCHING ALGORITHM

🧬 Fibonacci Search






Fibonacci Search uses a divide-and-conquer strategy utilizing Fibonacci numbers to pinpoint array target item lookups in a sorted collection array. It reduces the partition intervals mathematically without utilizing division operations.







#include <stdio.h>

int min(int x, int y) { return (x <= y) ? x : y; }

int fibonacciSearch(int a[], int n, int target) {
    int fibM2 = 0;
    int fibM1 = 1;
    int fibM = fibM2 + fibM1;
    
    while (fibM < n) {
        fibM2 = fibM1;
        fibM1 = fibM;
        fibM = fibM2 + fibM1;
    }
    
    int offset = -1;
    
    while (fibM > 1) {
        int i = min(offset + fibM2, n - 1);
        
        if (a[i] < target) {
            fibM = fibM1;
            fibM1 = fibM2;
            fibM2 = fibM - fibM1;
            offset = i;
        } else if (a[i] > target) {
            fibM = fibM2;
            fibM1 = fibM1 - fibM2;
            fibM2 = fibM - fibM1;
        } else {
            return i;
        }
    }
    
    if (fibM1 && a[offset + 1] == target) 
        return offset + 1;
        
    return -1;
}

int main() {
    int a[] = {10, 22, 35, 40, 45, 50, 80, 82, 85, 90, 100};
    int n = 11;
    int target = 85;
    printf("Found at index: %d\n", fibonacciSearch(a, n, target));
    return 0;
}















07 • ADVANCED LINKED STRUCT

💎 Skip List Search






Skip Lists build layered balance hierarchies across standard linked structures to approximate fast binary search times in linked lists. Below is a prototype layout implementation for tracking element lookup pathways.







#include <stdio.h>
#include <stdlib.h>

#define MAX_LEVEL 4

struct Node {
    int key;
    struct Node **forward;
};

struct SkipList {
    int level;
    struct Node *header;
};

struct Node* createNode(int key, int level) {
    struct Node *n = (struct Node*)malloc(sizeof(struct Node));
    n->key = key;
    n->forward = (struct Node**)malloc(sizeof(struct Node*) * (level + 1));
    for (int i = 0; i <= level; i++) n->forward[i] = NULL;
    return n;
}

int searchSkipList(struct SkipList *list, int target) {
    struct Node *current = list->header;
    for (int i = list->level; i >= 0; i--) {
        while (current->forward[i] != NULL && current->forward[i]->key < target) {
            current = current->forward[i];
        }
    }
    current = current->forward[0];
    if (current != NULL && current->key == target) return 1;
    return 0;
}

int main() {
    struct SkipList list;
    list.level = 0;
    list.header = createNode(-1, MAX_LEVEL);
    printf("Skip list structural components instantiated.\n");
    return 0;
}

09 • PLACEMENT PREPARATION

💼 Interview & Practice








🎯 Problem 1: Find First and Last Position of Element in Sorted Array


Given an array of integers sorted in non-decreasing order, find the starting and ending index of a given target value. If target is not found, return [-1, -1].









Hint: Run two separate binary searches. One modified to look for the first occurrence (move high to mid-1 when found) and one for the last occurrence (move low to mid+1 when found).









🎯 Problem 2: Search in a Sorted Array of Unknown Size


You are given a data stream array where you cannot access its size directly. Attempting to access indexes outside the boundary returns an error marker.









Hint: Start checking at index 1, and exponentially double the high boundary window bound (1, 2, 4, 8, 16...) until you discover an index that exceeds the target value or throws an exception. Then apply binary search between low and high.
