#include <stdio.h>

int main(void)
{
    int size, values[100];

    printf("Enter array size: ");
    scanf("%d", &size);
    if (size < 1 || size > 100) return 1;
    for (int index = 0; index < size; index++) scanf("%d", &values[index]);
    int last = values[size - 1];
    for (int index = size - 1; index > 0; index--) values[index] = values[index - 1];
    values[0] = last;
    for (int index = 0; index < size; index++) printf("%d%c", values[index], index == size - 1 ? '\n' : ' ');
    return 0;
}

PROGRAM EXPLANATION

Algorithm and explanation

  1. Read the required input values.
  2. Save the last element, shift the others right and place the saved value first.
  3. Display the computed result.
