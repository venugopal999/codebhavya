Convert Binary to Decimal

binary-to-decimal.c
Open in compiler
#include <stdio.h>

int main(void)
{
    char binary[65];
    unsigned long long decimal = 0;

    printf("Enter a binary number up to 64 bits: ");
    scanf("%64s", binary);
    for (int index = 0; binary[index] != '\0'; index++) {
        if (binary[index] != '0' && binary[index] != '1') {
            printf("Invalid binary number.\n");
            return 0;
        }
        decimal = decimal * 2 + (unsigned long long) (binary[index] - '0');
    }
    printf("Decimal = %llu\n", decimal);
    return 0;
}

PROGRAM EXPLANATION

Algorithm and explanation

  1. Read the required input values.
  2. Scan left to right, doubling the accumulated value before adding each bit.
  3. Display the computed result.
